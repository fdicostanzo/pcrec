# Optional trailing columns: flags that must hold (dead, look, red:S, ml).

LS+E	(?m)^\s+|\s+\z	trim, line start + absolute end
S+E	^#|//.*$	comment strip, non-m
LS+LE	(?m)^#|;\s*$	line start + line end
E+F	foo|\s+$	free + end
E+F	ab\z|cd	free + end, fixed width
S+F	^GET |POST	start + free

S+F	(?:^)?ab	redundant start	red:S
F	x(?:^|y)z	dead start path	dead
F	(?<=^)a	anchor in a lookbehind: unclassified	look
F	a(?=.*$)	anchor in a lookahead: unclassified	look
F	a(?m)^b	line anchor away from the edge is a landmark	ml
