=== A3: the RELATIVE forms, at several distances =====================
# (?+N) counts groups to the RIGHT of the call, (?-N) to the LEFT.
# The subjects are chosen so that the WRONG target gives a different
# answer, not merely a different capture.
  ^(a)(?-1)$                         'aa'         ((0, 2), [(0, 1)])                        # (?-1) -> group 1
  ^(a)(b)(?-1)$                      'abb'        ((0, 3), [(0, 1), (1, 2)])                # (?-1) -> group 2 (the NEAREST left)
  ^(a)(b)(?-2)$                      'aba'        ((0, 3), [(0, 1), (1, 2)])                # (?-2) -> group 1
  ^(?+1)(a)$                         'aa'         ((0, 2), [(1, 2)])                        # (?+1) -> group 1, a FORWARD call
  ^(?+2)(a)(b)$                      'bab'        ((0, 3), [(1, 2), (2, 3)])                # (?+2) -> group 2, forward past one
  ^(a)(?-01)$                        'aa'         ((0, 2), [(0, 1)])                        # leading zero in a relative call
  ^(a)\g<-1>$                        'aa'         ((0, 2), [(0, 1)])                        # \g<-N> is the same relative rule
  ^\g<+1>(a)$                        'aa'         ((0, 2), [(1, 2)])                        # \g<+N> forward
  ^(a)(b)(?2)$                       'abb'        ((0, 3), [(0, 1), (1, 2)])                # ABSOLUTE (?2) for contrast

# the FORWARD call is the one that makes a call unlike a reference:
# a forward REFERENCE can only ever be unset, a forward CALL works.

  ^\2(a|b)(c)$                       'abc'        None                                      # forward REFERENCE, for contrast
