{
    ptrdiff_t local[RX_NCAPS][2];   /* STEP writes NCAPS pairs whatever caps is */
    ptrdiff_t (*c)[2] = capture_spans ? capture_spans : local;
    size_t at;
    int r;
    if (__builtin_expect(it->hdr != PCREC_ITER_BOUND, 0))
        if (!rx__enter(it)) return 0;                  /* bind; or refuse; or ENDED */
    r = rx__step(it, c, &at);
    if (r == 1) return 1;
    it->lo = at;                                       /* the search that ended it */
    it->rc = r;
    it->hdr = PCREC_ITER_ENDED;                        /* sticky */
    return 0;
}
